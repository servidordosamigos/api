        <!-- Caixa de texto onde o resultado vai aparecer -->
        <textarea id="caixaResultado" placeholder="O resultado da API aparecerá aqui..." readonly></textarea>

        <!-- Botão para copiar o conteúdo -->
        <button class="btn-copiar" onclick="copiarTexto()">Copiar Texto</button>

    <script>
        async function gerarTexto() {
            const caixa = document.getElementById('caixaResultado');
            caixa.value = "A carregar...";

            try {
                // Faz o pedido para a sua API na Vercel
                const resposta = await fetch('/api/mensagem', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ texto: 'gerar conteúdo' })
                });

                const dados = await resposta.json();
                caixa.value = dados.resultado;
            } catch (erro) {
                caixa.value = "Erro ao comunicar com a API.";
                console.error(erro);
            }
        }

        function copiarTexto() {
            const caixa = document.getElementById('caixaResultado');
            
            if (!caixa.value) return;

            caixa.select();
            caixa.setSelectionRange(0, 99999);
            navigator.clipboard.writeText(caixa.value);

            const botaoCopiar = event.target;
            const textoOriginal = botaoCopiar.innerText;
            botaoCopiar.innerText = "Copiado!";
            botaoCopiar.style.backgroundColor = "#155724";

            setTimeout(() => {
                botaoCopiar.innerText = textoOriginal;
                botaoCopiar.style.backgroundColor = "#28a745";
            }, 2000);
        }
    </script>
