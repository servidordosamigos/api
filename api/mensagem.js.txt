export default function handler(req, res) {
    // Permite que qualquer site (incluindo o Google Sites) aceda a esta API
    res.setHeader('Access-Control-Allow-Origin', '*');
    res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS');
    res.setHeader('Access-Control-Allow-Headers', 'Content-Type');

    // Se for um pedido de verificação do navegador, responde logo com sucesso
    if (req.method === 'OPTIONS') {
        return res.status(200).end();
    }

    if (req.method !== 'POST') {
        return res.status(405).json({ erro: 'Apenas pedidos POST são permitidos' });
    }

    const { texto } = req.body || {};

    const textoProcessado = texto ? texto.toUpperCase() : 'OLA MUNDO';

    return res.status(200).json({ 
        sucesso: true,
        resultado: textoProcessado,
        mensagem: 'API a funcionar perfeitamente!'
    });
}
